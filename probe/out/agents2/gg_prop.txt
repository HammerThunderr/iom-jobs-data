                      <h2 id="summary">Summary</h2>
            <div class="pink-links"><p><p>Situated in the heart of Castletown, the Island’s historic capital, this attractive stone-fronted period home offers a wonderful blend of traditional character and stylish contemporary living. Beautifully presented throughout, the property combines charming period features with high-quality modern fixtures and fittings, creating a warm and inviting home in an exceptionally convenient location. Local shops, restaurants, the beach, swimming pool and schools are all within easy walking distance, making this an ideal home for those wishing to enjoy the very best of village and coastal living. Offered for sale with no onward chain.</p>
<p>The accommodation is introduced by an elegant living room, providing a lovely space in which to relax. Bi-folding doors open through to the separate dining room, creating an excellent connection between the reception spaces while retaining the flexibility to enjoy each room independently. The dining room features an attractive fireplace, adding further character and charm, and provides access to the rear porch. A modern breakfast kitchen completes the principal living accommodation, offering a stylish and practical space for everyday dining and entertaining.</p>
<p>The upper floors provide four bedrooms, including a truly impressive principal bedroom which is a particular highlight of the home. Beautifully appointed, this stunning room features an exposed stone wall, vaulted ceiling with exposed beams and distant rural views, creating a wonderfully characterful and peaceful retreat. The principal bedroom is further complemented by a luxurious en-suite shower room, while a separate stylish family shower room serves the remaining bedrooms.</p>

  var offer_email = '';
  $(document).ready(function() {

    let ViewingTimes = new Timeslots('#inlineBookingForm', {slotType: 'viewing'});

    function loadBookingModal(address, price, photo, confirm) {

      $('.warnMessage_view').html('').hide();
      triggerChange = false;

      let $modal      = $('#bookViewingModal');

      
      $modal.find('.address').html(address);
      $modal.find('.price').html(price);
      if(photo.trim() !== '')
        $modal.find('.photo').attr('src', photo);
      else
        $modal.find('.photo').attr('src', '/assets/images/logo.svg');

      $modal.find('.submit_booking').unbind('click').click(function() {
        $modal.scrollTop(250);
        let bookingInfo = {};
        $modal.find(".bookForm :input").not('button').each(function () {
          if ($(this).attr("name").length > 0) {
            if($(this).is(':checkbox')) {
              if($(this).is(':checked')) {
                bookingInfo[$(this).attr("name")] = 1;
              } else {
                bookingInfo[$(this).attr("name")] = 0;
              }
            } else if($(this).is(':radio')) {
              bookingInfo[$(this).attr("name")] = $modal.find('.bookForm input[name="'+$(this).attr("name")+'"]:checked').val();
            } else
              bookingInfo[$(this).attr("name")] = $(this).val();
          }
        });
        validateInput(bookingInfo, confirm);
      });

    }

    function validateInput(bookingInfo, confirm) {
      $('.warnMessage_view').html('').hide();
      $('.bookForm .loader').show();
      let valid = true;
      let errorMsg = 'Please provide the following information:<br />';
      let checkFields = [
        {field: 'buying_status', niceName: 'Buying status'},
        {field: 'mortgage_or_cash_offer', niceName: 'Mortgage or cash offer'},
        {field: 'sms_opt_in', niceName: 'SMS opt in'},
        {field: 'email', niceName:'Email'},
        {field: 'mobile', niceName:'Mobile number'}
      ];

      if (bookingInfo['full_name'].trim() === '') {
        valid = false;
        errorMsg += '* Full name<br />';
      }

      for (let i = 0; i < checkFields.length; i++) {
        if (typeof bookingInfo[checkFields[i].field] === 'undefined') {

                valid = false;
                errorMsg += '* Email address is not valid<br />';
              }

              if(resultData.data.result.email === 'valid') {
                let email = bookingInfo['email'];
                let alias = email.split('@')[0];
                let domain = email.split('@')[1];
                if(domain === 'googlemail.com') {
                  valid = false;
                  errorMsg += '* Please register using '+alias+'@gmail.com instead of '+alias+'@googlemail.com as the latter format is not supported<br />';
                }
              }

              if (valid) {
                $('.warnMessage_view').html('').hide();

                let date = ViewingTimes.date;
                let agent = ViewingTimes.agent;
                let property = ViewingTimes.property;
                let offerMade = ViewingTimes.offerMade;

    function getCaret(el) {
      if (el.selectionStart) {
        return el.selectionStart;
      } else if (document.selection) {
        el.focus();

        var r = document.selection.createRange();
        if (r == null) {
          return 0;
        }

        var re = el.createTextRange(),
          rc = re.duplicate();
        re.moveToBookmark(r.getBookmark());
        rc.setEndPoint('EndToStart', re);

        return rc.text.length;
      }
      return 0;
    }

  function addCommas(nStr)
  {
    nStr += '';
    x = nStr.split('.');
    x1 = x[0];
    x2 = x.length > 1 ? '.' + x[1] : '';
    var rgx = /(\d+)(\d{3})/;
    while (rgx.test(x1)) {
      x1 = x1.replace(rgx, '$1' + ',' + '$2');
    }
    return '£' + x1 + x2;
  }

  var initPhotoSwipe = function(gallerySelector) {

    var parseThumbnailElements = function(el) {
      var thumbElements = el.childNodes,
        numNodes = thumbElements.length,
        items = [],
        figureEl,
        linkEl,
        size,
        item;

      for(var i = 0; i < numNodes; i++) {

        figureEl = thumbElements[i];
        if(figureEl.nodeType !== 1) {
          continue;
        }

        linkEl = figureEl.children[0];
        size = linkEl.getAttribute('data-size').split('x');
        item = {
          src: linkEl.getAttribute('href'),
          w: parseInt(size[0], 10),
          h: parseInt(size[1], 10)
        };

        if(linkEl.getAttribute('data-title')) {
          item.title = linkEl.getAttribute('data-title');
        }

        if(linkEl.children.length > 0) {
          item.msrc = linkEl.children[0].getAttribute('src');
        }

        item.el = figureEl;
        items.push(item);

      }

      return items;

    };

    var closest = function closest(el, fn) {
      return el && ( fn(el) ? el : closest(el.parentNode, fn) );
    };

    var onThumbnailsClick = function(e) {

      e = e || window.event;
      e.preventDefault ? e.preventDefault() : e.returnValue = false;
      var eTarget = e.target || e.srcElement;

      var clickedListItem = closest(eTarget, function(el) {
        return (el.tagName && el.tagName.toUpperCase() === 'LI');
      });

      if(!clickedListItem) {
        return;
      }

      var clickedGallery = clickedListItem.parentNode,
        childNodes = clickedListItem.parentNode.childNodes,
        numChildNodes = childNodes.length,
        nodeIndex = 0,
        index;

      for(var i = 0; i < numChildNodes; i++) {
        if(childNodes[i].nodeType !== 1) {
          continue;
        }
        if(childNodes[i] === clickedListItem) {
          index = nodeIndex;
          break;
        }
        nodeIndex++;
      }

      if(index >= 0) {
        openPhotoSwipe(index, clickedGallery);
      }

      return false;

    };

    var openPhotoSwipe = function(index, galleryElement) {

      var pswpElement = document.querySelectorAll('.pswp')[0],
        gallery,
        options,
        items;

      items = parseThumbnailElements(galleryElement);
      options = {
        galleryUID: galleryElement.getAttribute('data-pswp-uid'),
        getThumbBoundsFn: function(index) {
          var thumbnail = items[index].el.getElementsByTagName('img')[0],
            pageYScroll = window.pageYOffset || document.documentElement.scrollTop,
            rect = thumbnail.getBoundingClientRect();
          return {x:rect.left, y:rect.top + pageYScroll, w:rect.width};
        }
      };

      options.index = parseInt(index, 10);

      if(isNaN(options.index)) {
        return;
      }

      gallery = new PhotoSwipe( pswpElement, PhotoSwipeUI_Default, items, options);
      gallery.init();

    };

    var galleryElements = document.querySelectorAll(gallerySelector);
    for(var i = 0, l = galleryElements.length; i < l; i++) {
      galleryElements[i].setAttribute('data-pswp-uid', i+1);
      galleryElements[i].onclick = onThumbnailsClick;
    }

  };
