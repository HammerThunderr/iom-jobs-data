var finish = false;
var ready = true; //Assign the flag here

function load_results(){
    if($('#properties .loader').length){ return false;}
    var page = jQuery('#page').val();
    $('#properties').append('<div class="loader"></div>');
    $.ajax({
        url:firstURL,
        method:'POST',
        data:$( "#FilterForm" ).serialize(),
        success: function(html) { // this gets called multiple times on erratic scrolling
            if($('#page').val()==1){$('#properties').html('')} // blank out the table
            if ($.trim(html)){
                $('#properties').append(html);
                $('#page').val(parseInt($('#page').val()) + 1);
            }else{
                finish = true; // Stops repeat calls when we reach the end
                if($('#page').val()==1){
                $('#properties').append('<div class="loaded">Sorry, no results found, try widening your search criteria, if you create a property alert we will automatically email you when we have a match.</div>');
                }else{
                $('#properties').append('<div class="loaded">End of Results</div>');
                }

            }
            $('#properties .loader').remove();
            $("#FilterForm").trigger("sticky_kit:recalc");
        }
    }).always(function(){
        ready = true; //Reset the flag here
    });
}

function append_query(){
if (history.replaceState) {

var query = '';
    $('#FilterForm .form-control').each(function(){
        if(query.indexOf("?") === -1){ query += '?'; }else{ query += '&'; }
        var name = $(this).attr('name');
        var value = $(this).val();
        query += name + '=' + value;
    });
    var newUrl = document.location.protocol +"//"+ document.location.hostname + document.location.pathname + query;
    window.history.pushState({path:newUrl},'',newUrl);
}
}

        ready = false; //Set the flag here
        if (firstURL !== null) {

    }
