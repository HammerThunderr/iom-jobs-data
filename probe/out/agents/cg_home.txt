        const requiredFields = ['first_name', 'email', 'phone'];
        let isValid = true;

        // Clear previous error states and restore original placeholders
        form.find('input').each(function(){
            const $inp = $(this);
            if ($inp.data('origPlaceholder') !== undefined) {
                $inp.attr('placeholder', $inp.data('origPlaceholder'));
            }

            const val = input.val();
            if (!val || !val.toString().trim()) {

                }

            }
        });

    function onSubmit(token) {
        var form = $('#ContactForm');
        var url = form.attr('action');

        if (!validateHomeValuationForm()) {
            try { grecaptcha.reset(); } catch (e) {}
            $('.g-recaptcha').prop('disabled', false);
            return false;
        }

        // Prevent multiple submissions
        if (form.data('submitting')) { return false; }
        form.data('submitting', true);
        var $btn = form.find('.g-recaptcha');
        $btn.prop('disabled', true).addClass('disabled');
        form.addClass('is-submitting').animate({ opacity: 0.5 }, 200);

        $.ajax({
            type: "POST",
            url: url,
            data: form.serialize(),
            success: function(data) {
                if(data == 'true'){
                    $('#ContactForm').html('<div class=\"alert alert-success\"><h2>Thank you, your message has been sent</h2></div>');
                }else{
                    $('#ContactForm').append('<div class=\"alert alert-danger\"><h2>Sorry, something went wrong</h2></div>');
                    form.removeClass('is-submitting').animate({ opacity: 1 }, 200);
                    $btn.prop('disabled', false).removeClass('disabled');
                    form.data('submitting', false);
                    try { grecaptcha.reset(); } catch (e) {}
                }
            },
            error: function(){
                $('#ContactForm').append('<div class=\"alert alert-danger\"><h2>Sorry, something went wrong</h2></div>');
                form.removeClass('is-submitting').animate({ opacity: 1 }, 200);
                $btn.prop('disabled', false).removeClass('disabled');
                form.data('submitting', false);
                try { grecaptcha.reset(); } catch (e) {}
            }
        });
    }
</script>

<script>
    const animationDuration = 1500;
    const frameDuration = 1000 / 60;
    const totalFrames = Math.round( animationDuration / frameDuration );
    const easeOutQuad = t => t * ( 2 - t );
    const animateCountUp = el => {
        let frame = 0;
        const countTo = parseInt( el.innerHTML, 10 );
        const counter = setInterval( () => {
            frame++;
            const progress = easeOutQuad( frame / totalFrames );
            const currentCount = Math.round( countTo * progress );
            if ( parseInt( el.innerHTML, 10 ) !== currentCount ) { el.innerHTML = currentCount.toLocaleString(); }
            if ( frame === totalFrames ) { clearInterval( counter ); }
        }, frameDuration );
    };
    const runAnimations = () => {
        jQuery('#stats').addClass('in');
        const countupEls = document.querySelectorAll('.countup');
        countupEls.forEach( animateCountUp );
    };


    function isInViewport(el) {
        const rect = el.getBoundingClientRect();
        return (rect.top >= 0 && rect.left >= 0 && rect.bottom <= (window.innerHeight || document.documentElement.clientHeight) && rect.right <= (window.innerWidth || document.documentElement.clientWidth));
    }
    const box = document.querySelector('#stats');
    document.addEventListener('scroll', function () {if(isInViewport(box) && !jQuery('#stats').hasClass('in')){ runAnimations(); console.log('in'); }}, {passive: true});

</script>
